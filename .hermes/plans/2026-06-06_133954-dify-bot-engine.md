```typescript
import type { Knex } from 'knex';
import type { GoldLogger } from '../../core/logger.js';

export interface DifyBotConfig {
  id: string;
  account_id: string;
  name: string;
  dify_api_key: string;
  dify_webhook_url: string;
  enabled: boolean;
  filter_mode: 'all' | 'keywords' | 'mention';
  filter_keywords: string[];
  created_at: string;
  updated_at: string;
}

export class DifyBotService {
  constructor(
    private readonly knex: Knex,
    private readonly logger: GoldLogger,
  ) {}

  async listBots(accountId?: string): Promise<DifyBotConfig[]> {
    let query = this.knex<DifyBotConfig>('dify_bots').orderBy('created_at', 'desc');
    if (accountId) query = query.where('account_id', accountId);
    return query;
  }

  async getBot(id: string): Promise<DifyBotConfig | undefined> {
    return this.knex<DifyBotConfig>('dify_bots').where('id', id).first();
  }

  async getEnabledBotsForAccount(accountId: string): Promise<DifyBotConfig[]> {
    return this.knex<DifyBotConfig>('dify_bots')
      .where('account_id', accountId)
      .where('enabled', true);
  }

  async createBot(data: Omit<DifyBotConfig, 'id' | 'created_at' | 'updated_at'>): Promise<DifyBotConfig> {
    const [bot] = await this.knex<DifyBotConfig>('dify_bots').insert(data).returning('*');
    return bot;
  }

  async updateBot(id: string, data: Partial<Omit<DifyBotConfig, 'id' | 'created_at'>>): Promise<DifyBotConfig | undefined> {
    const [bot] = await this.knex<DifyBotConfig>('dify_bots')
      .where('id', id)
      .update({ ...data, updated_at: this.knex.fn.now() })
      .returning('*');
    return bot;
  }

  async deleteBot(id: string): Promise<boolean> {
    const count = await this.knex('dify_bots').where('id', id).delete();
    return count > 0;
  }
}
```

### Task 3: DifyBotExecutor — gọi Dify webhook + gửi reply

**Objective:** Core bot logic: nhận message → gọi Dify → gửi reply

**Files:**
- Create: `backend/src/server/services/dify-bot-executor.ts`

**Code:**

```typescript
import type { DifyBotService, DifyBotConfig } from './dify-bot-service.js';
import type { GoldLogger } from '../../core/logger.js';
import type { GoldRuntime } from '../../core/runtime.js';

interface IncomingMessage {
  id: string;
  conversationId: string;
  threadId: string;
  text: string;
  senderId?: string;
  senderName?: string;
  isSelf: boolean;
  timestamp: string;
}

export class DifyBotExecutor {
  constructor(
    private readonly botService: DifyBotService,
    private readonly logger: GoldLogger,
  ) {}

  /** Returns true if the message matches the bot's filter rules */
  private matchesFilter(bot: DifyBotConfig, msg: IncomingMessage): boolean {
    if (!bot.enabled) return false;
    if (msg.isSelf) return false; // không reply chính mình

    switch (bot.filter_mode) {
      case 'all':
        return true;
      case 'keywords': {
        const keywords: string[] = bot.filter_keywords ?? [];
        const lower = msg.text.toLowerCase();
        return keywords.some((kw) => lower.includes(kw.toLowerCase()));
      }
      case 'mention':
        // mention check will need account display name — simplified for now
        return true;
      default:
        return false;
    }
  }

  async processMessage(
    accountId: string,
    runtime: GoldRuntime,
    message: IncomingMessage,
  ): Promise<void> {
    const bots = await this.botService.getEnabledBotsForAccount(accountId);
    if (bots.length === 0) return;

    for (const bot of bots) {
      if (!this.matchesFilter(bot, message)) continue;

      try {
        this.logger.info('dify_bot_processing', { botId: bot.id, messageId: message.id });

        const response = await fetch(bot.dify_webhook_url, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${bot.dify_api_key}`,
          },
          body: JSON.stringify({
            text: message.text,
            sender_name: message.senderName ?? message.senderId ?? 'unknown',
            sender_id: message.senderId,
            conversation_id: message.conversationId,
            account_id: accountId,
            message_id: message.id,
          }),
        });

        if (!response.ok) {
          this.logger.error('dify_bot_webhook_error', {
            botId: bot.id,
            status: response.status,
            statusText: response.statusText,
          });
          continue;
        }

        const data = await response.json() as { text?: string; answer?: string; output?: string };
        const replyText = data.text ?? data.answer ?? data.output;
        if (replyText && replyText.trim()) {
          await runtime.sendMessage(message.conversationId, replyText.trim());
          this.logger.info('dify_bot_reply_sent', { botId: bot.id, messageId: message.id });
        }
      } catch (err) {
        this.logger.error('dify_bot_error', {
          botId: bot.id,
          error: err instanceof Error ? err.message : String(err),
        });
      }
    }
  }
}
```

```typescript
private difyBotExecutor?: DifyBotExecutor;

setDifyBotExecutor(executor: DifyBotExecutor) {
  this.difyBotExecutor = executor;
}

// Trong method wireRuntime (hoặc method attach listener hiện có):
private wireBotExecutor(runtime: GoldRuntime, accountId: string) {
  if (!this.difyBotExecutor) return;
  runtime.listener.onConversationMessage((msg) => {
    if (msg.isSelf) return;
    this.difyBotExecutor!.processMessage(accountId, runtime, {
      id: msg.id,
      conversationId: msg.conversationId,
      threadId: msg.threadId,
      text: msg.text,
      senderId: msg.senderId,
      senderName: msg.senderName,
      isSelf: msg.isSelf,
      timestamp: msg.timestamp,
    }).catch(err => this.logger.error('bot_executor_wire_error', { error: String(err) }));
  });
}
```

### Task 5: API Routes cho Bot Admin

**Objective:** REST API CRUD cho bot configs

**Files:**
- Create: `backend/src/server/routes/dify-bots.ts`

**Code:**

```typescript
import { Router, type Request, type Response } from 'express';
import type { DifyBotService } from '../services/dify-bot-service.js';

export function createDifyBotsRouter(
  botService: DifyBotService,
  requireAuth: (req: Request, res: Response, next: Function) => void,
  requireAdmin: (req: Request, res: Response, next: Function) => void,
) {
  const router = Router();
  router.use(requireAuth);

  router.get('/', (req, res) => {
    const accountId = req.query.accountId as string | undefined;
    botService.listBots(accountId).then((bots) => res.json({ bots })).catch((err) => res.status(500).json({ error: err.message }));
  });

  router.post('/', requireAdmin, (req, res) => {
    botService.createBot(req.body).then((bot) => res.json(bot)).catch((err) => res.status(400).json({ error: err.message }));
  });

  router.put('/:id', requireAdmin, (req, res) => {
    botService.updateBot(req.params.id, req.body).then((bot) => {
      if (!bot) { res.status(404).json({ error: 'Bot not found' }); return; }
      res.json(bot);
    }).catch((err) => res.status(400).json({ error: err.message }));
  });

  router.delete('/:id', requireAdmin, (req, res) => {
    botService.deleteBot(req.params.id).then((ok) => {
      if (!ok) { res.status(404).json({ error: 'Bot not found' }); return; }
      res.json({ ok: true });
    }).catch((err) => res.status(500).json({ error: err.message }));
  });

  return router;
}
```
